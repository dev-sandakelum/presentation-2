    36→    36→/**
    37→    37→ * Format a TSX buffer once, when it first appears in the editor
    38→    38→ * (fix-list item 2). The result is applied through onBufferChange, so it
    39→    39→ * is a single undoable edit that autosaves like any other change.
    40→    40→ */
    41→    41→function useFormatOnOpen(
    42→    42→  path: string | null,
    43→    43→  content: string | undefined,
    44→    44→  apply: (formatted: string) => void,
    45→    45→) {
    46→    46→  const doneRef = useRef<Set<string>>(new Set());
    47→    47→  const busyRef = useRef<string | null>(null);
    48→    48→  useEffect(() => {
    49→    49→    if (!path || content === undefined) return;
    50→    50→    if (!FORMATABLE_EXT.test(path) || doneRef.current.has(path) || busyRef.current === path) return;
    51→    51→    busyRef.current = path;
    52→    52→    let cancelled = false;
    53→    53→    (async () => {
    54→    54→      const formatted = await formatTsx(content);
    55→    55→      if (formatted !== content) apply(formatted);
    56→    56→      if (!cancelled) doneRef.current.add(path);
    57→    57→      busyRef.current = null;
    58→    58→    })();
    59→    59→    return () => {
    60→    60→      cancelled = true;
    61→    61→    };
    62→    62→  }, [path, content, apply]);
    63→    63→}

   389→   389→function SourceEditor(props: {
   390→   390→  layer: "bg" | "overlay";
   391→   391→  deckId: string;
   392→   392→  deck: DeckJson;
   393→   393→  files: FileRow[];
   394→   394→  sel: SlideEntry | null;
   395→   395→  buffers: Record<string, string>;
   396→   396→  onBufferChange: (path: string, content: string) => void;
   397→   397→  ensureBuffer: (path: string) => Promise<void>;
   398→   398→  onCommonSource: (layer: "bg" | "overlay", source: Source) => void;
   399→   399→  onOverride: (slideId: string, layer: "bg" | "overlay", source: Source | undefined) => void;
   400→   400→  activeFileRef: React.RefObject<string | null>;
   401→   401→  formatFile: (path: string | null) => Promise<void>;
   402→   402→  cmTheme: "dark" | "light";
   403→   403→  pasteFormatExt: unknown;
   404→   404→}) {
   405→   405→  const { layer, deckId, deck, files, sel } = props;
   406→   406→  const { toast } = useToast();
   407→   407→  const [scope, setScope] = useState<"common" | "override">("common");
   408→   408→
   409→   409→  const common = layer === "bg" ? deck.bg : deck.overlay;
   410→   410→  const override = sel ? (layer === "bg" ? sel.bg : sel.overlay) : undefined;
   411→   411→  const effective = scope === "override" ? override : common;
   412→   412→  const dirFiles = files
   413→   413→    .filter((f) => f.path.startsWith(`${layer}/`) && /\.(tsx|ts)$/.test(f.path))
   414→   414→    .map((f) => f.path);
   415→   415→  const images = files.filter((f) => f.path.startsWith("assets/") && /\.(png|jpe?g|svg|webp|gif|avif)$/i.test(f.path)).map((f) => f.path);
   416→   416→
   417→   417→  const codePath = effective && effective.type === "code" ? effective.path : null;
   418→   418→
   419→   419→  useEffect(() => {
   420→   420→    if (codePath) props.ensureBuffer(codePath);
   421→   421→  }, [codePath]);
   422→   422→
   423→   423→  // register the visible file with the pane-level formatter
   424→   424→  const activeFileRef = props.activeFileRef;
   425→   425→  useEffect(() => {
   426→   426→    activeFileRef.current = codePath;
   427→   427→    return () => {
   428→   428→      if (activeFileRef.current === codePath) activeFileRef.current = null;
   429→   429→    };
   430→   430→  }, [codePath, activeFileRef]);
   431→   431→
   432→   432→  const applyFormat = useCallback(
   433→   433→    (formatted: string) => {
   434→   434→      if (codePath) props.onBufferChange(codePath, formatted);
   435→   435→    },
   436→   436→    [codePath],
   437→   437→  );
   438→   438→  useFormatOnOpen(codePath, codePath ? props.buffers[codePath] : undefined, applyFormat);
   439→   439→
   440→   440→  const setSource = (source: Source | undefined) => {
   441→   441→    if (scope === "override" && sel) {
   442→   442→      props.onOverride(sel.id, layer, source);
   443→   443→    } else {
   444→   444→      props.onCommonSource(layer, source ?? { type: "none" });
   445→   445→    }
   446→   446→  };

    36→/**
    37→ * Format a TSX buffer once, when it first appears in the editor
    38→ * (fix-list item 2). The result is applied through onBufferChange, so it
    39→ * is a single undoable edit that autosaves like any other change.
    40→ */
    41→function useFormatOnOpen(
    42→  path: string | null,
    43→  content: string | undefined,
    44→  apply: (formatted: string) => void,
    45→) {
    46→  const doneRef = useRef<Set<string>>(new Set());
    47→  const busyRef = useRef<string | null>(null);
    48→  useEffect(() => {
    49→    if (!path || content === undefined) return;
    50→    if (!FORMATABLE_EXT.test(path) || doneRef.current.has(path) || busyRef.current === path) return;
    51→    busyRef.current = path;
    52→    let cancelled = false;
    53→    (async () => {
    54→      const formatted = await formatTsx(content);
    55→      if (formatted !== content) apply(formatted);
    56→      if (!cancelled) doneRef.current.add(path);
    57→      busyRef.current = null;
    58→    })();
    59→    return () => {
    60→      cancelled = true;
    61→    };
    62→  }, [path, content, apply]);
    63→}

    98→  changeRef.current = onBufferChange;

   100→  /** the file currently shown in the editor (slide / bg / overlay) */

   111→    async (path: string | null) => {

   120→      if (formatted !== content) {
   121→        changeRef.current(path, formatted);
   122→        toast({ title: "Formatted", description: path });
   123→      }
   124→    },
   125→    [toast],
   126→  );
   127→  const formatActive = useCallback(() => formatFile(activeFileRef.current), [formatFile]);
   128→
   129→  // Shift+Alt+F (fix-list item 2)
   130→  useEffect(() => {
   131→    const onKey = (e: KeyboardEvent) => {
   132→      if (e.altKey && e.shiftKey && (e.key === "f" || e.key === "F")) {
   133→        e.preventDefault();
   134→        formatActive();
   135→      }
   136→    };
   137→    window.addEventListener("keydown", onKey);
   138→    return () => window.removeEventListener("keydown", onKey);

   140→
   141→  // format-on-open for the slide file

   154→      : tab === "theme" ? "theme.css"

   189→      const { EditorSelection, EditorView: EV } = await import("./editor/bits");

   195→        selection: EditorSelection.cursor(pos),

   201→      cancelled = true;
   202→    };

   222→        }),
   223→      );
   224→    });

   226→      cancelled = true;
   227→    };

   231→    { key: "slide", label: "Slide", badge: Boolean(slideError) },
   232→    { key: "bg", label: "BG" },
   233→    { key: "overlay", label: "Overlay" },
   234→    { key: "notes", label: "Notes" },
   235→    { key: "assets", label: "Assets" },
   236→    { key: "theme", label: "Theme" },
   237→  ];
   238→
   239→  const cmTheme: "dark" | "light" = uiTheme === "light" ? "light" : "dark";
   240→
   241→  return (
   242→    <section className={`ss-codepane ${props.minimized ? "min" : ""}`}>
   243→      <div className="ss-codepane-head">
   244→        <div className="ss-code-tabs">
   245→          {tabs.map((t) => (
   246→            <button key={t.key} className={`ss-code-tab ${tab === t.key ? "on" : ""}`} onClick={() => props.onTab(t.key)}>
   247→              {t.label}
   248→              {t.badge && <span className="badge">!</span>}
   249→            </button>
   250→          ))}
   251→        </div>
   252→        {!props.minimized && (
   253→          <>
   254→            <div style={{ flex: 1 }} />
   255→            {tab === "slide" && sel && (
   256→              <span className="ss-label" style={{ letterSpacing: ".06em" }}>{slidePath}</span>
   257→            )}
   258→            {tab === "notes" && sel && <span className="ss-label">{notesPath}</span>}
   259→            {tab === "theme" && <span className="ss-label">theme.css</span>}
   260→            {slideError && tab === "slide" && (
   261→              <button className="ss-btn" style={{ height: 26, padding: "0 10px", fontSize: 12 }} onClick={copyFix} title="Copy slide code + error as an AI fix prompt">
   262→                <WandSparkles size={13} /> Copy fix prompt
   263→              </button>
   264→            )}
   265→            {(tab === "slide" || tab === "bg" || tab === "overlay") && (
   266→              <button className="ss-btn" style={{ height: 26, padding: "0 10px", fontSize: 12 }} onClick={() => formatActive()} title="Format file (Shift+Alt+F)">
   267→                <Braces size={13} /> Format
   268→              </button>

   270→            <button className="ss-btn" style={{ height: 26, padding: "0 10px", fontSize: 12 }} onClick={props.onFlushSave} title="Save now (Ctrl+S)">
   271→              <Save size={13} /> Save

   275→        {props.minimized && <div style={{ flex: 1 }} />}

   277→          className="ss-btn ss-btn-icon"
   278→          style={{ width: 28, height: 26, flex: "none" }}
   279→          onClick={props.onToggleMinimize}
   280→          title={props.minimized ? "Restore code pane (Ctrl+E)" : "Minimize code pane (Ctrl+E)"}
   281→        >
   282→          {props.minimized ? <ChevronsUp size={14} /> : <ChevronsDown size={14} />}
   283→        </button>
   284→      </div>
   285→
   286→      {!props.minimized && (
   287→        <>
   288→          {slideError && (tab === "slide" || tab === "bg" || tab === "overlay") && (
   289→            <div className="ss-error-banner">
   290→              <span style={{ color: "var(--ss-err)", flex: "none", fontWeight: 600 }}>
   291→                <AlertTriangle size={12} style={{ verticalAlign: -1, marginRight: 6 }} />
   292→                Error
   293→              </span>
   294→              <span>{slideError}</span>
   295→            </div>
   296→          )}
   297→
   298→          <div className="ss-codepane-body ss-cm">
   299→            {tab === "slide" && sel && (
   300→              <CmEditor
   301→                value={buffers[slidePath] ?? ""}
   302→                theme={cmTheme}
   303→                lang="tsx"
   304→                pasteFormatExt={pasteFormatExt}
   305→                onChange={(v) => onBufferChange(slidePath, v)}
   306→                onCreateEditor={(view) => {
   307→                  viewRef.current = view;
   308→                }}
   309→              />
   310→            )}
   311→            {tab === "slide" && !sel && <EmptyPane text="Select a slide in the filmstrip" />}
   312→
   313→            {tab === "bg" && (
   314→              <SourceEditor
   315→                layer="bg"
   316→                deckId={deckId}
   317→                deck={deck}
   318→                files={props.files}
   319→                sel={sel}
   320→                buffers={buffers}
   321→                onBufferChange={onBufferChange}
   322→                ensureBuffer={ensureBuffer}
   323→                onCommonSource={props.onCommonSource}
   324→                onOverride={props.onOverride}
   325→                activeFileRef={activeFileRef}
   326→                formatFile={formatFile}
   327→                cmTheme={cmTheme}
   328→                pasteFormatExt={pasteFormatExt}
   329→              />
   330→            )}
   331→            {tab === "overlay" && (
   332→              <SourceEditor
   333→                layer="overlay"
   334→                deckId={deckId}
   335→                deck={deck}
   336→                files={props.files}
   337→                sel={sel}
   338→                buffers={buffers}
   339→                onBufferChange={onBufferChange}
   340→                ensureBuffer={ensureBuffer}
   341→                onCommonSource={props.onCommonSource}
   342→                onOverride={props.onOverride}
   343→                activeFileRef={activeFileRef}
   344→                formatFile={formatFile}
   345→                cmTheme={cmTheme}
   346→                pasteFormatExt={pasteFormatExt}

   350→            {tab === "notes" && sel && (

   367→                onChange={(v) => onBufferChange("theme.css", v)}

   369→            )}
   370→          </div>
   371→        </>
   372→      )}
   373→    </section>
   374→  );
   375→});

   388→
   389→function SourceEditor(props: {
   390→  layer: "bg" | "overlay";
   391→  deckId: string;
   392→  deck: DeckJson;
   393→  files: FileRow[];
   394→  sel: SlideEntry | null;
   395→  buffers: Record<string, string>;
   396→  onBufferChange: (path: string, content: string) => void;
   397→  ensureBuffer: (path: string) => Promise<void>;
   398→  onCommonSource: (layer: "bg" | "overlay", source: Source) => void;
   399→  onOverride: (slideId: string, layer: "bg" | "overlay", source: Source | undefined) => void;
   400→  activeFileRef: React.RefObject<string | null>;
   401→  formatFile: (path: string | null) => Promise<void>;
   402→  cmTheme: "dark" | "light";
   403→  pasteFormatExt: unknown;
   404→}) {
   405→  const { layer, deckId, deck, files, sel } = props;
   406→  const { toast } = useToast();
   407→  const [scope, setScope] = useState<"common" | "override">("common");
   408→
   409→  const common = layer === "bg" ? deck.bg : deck.overlay;
   410→  const override = sel ? (layer === "bg" ? sel.bg : sel.overlay) : undefined;
   411→  const effective = scope === "override" ? override : common;
   412→  const dirFiles = files
   413→    .filter((f) => f.path.startsWith(`${layer}/`) && /\.(tsx|ts)$/.test(f.path))
   414→    .map((f) => f.path);
   415→  const images = files.filter((f) => f.path.startsWith("assets/") && /\.(png|jpe?g|svg|webp|gif|avif)$/i.test(f.path)).map((f) => f.path);
   416→
   417→  const codePath = effective && effective.type === "code" ? effective.path : null;
   418→
   419→  useEffect(() => {
   420→    if (codePath) props.ensureBuffer(codePath);
   421→  }, [codePath]);
   422→
   423→  // register the visible file with the pane-level formatter
   424→  const activeFileRef = props.activeFileRef;
   425→  useEffect(() => {
   426→    activeFileRef.current = codePath;
   427→    return () => {
   428→      if (activeFileRef.current === codePath) activeFileRef.current = null;
   429→    };
   430→  }, [codePath, activeFileRef]);
   431→
   432→  const applyFormat = useCallback(
   433→    (formatted: string) => {
   434→      if (codePath) props.onBufferChange(codePath, formatted);
   435→    },
   436→    [codePath],
   437→  );
   438→  useFormatOnOpen(codePath, codePath ? props.buffers[codePath] : undefined, applyFormat);
   439→
   440→  const setSource = (source: Source | undefined) => {
   441→    if (scope === "override" && sel) {
   442→      props.onOverride(sel.id, layer, source);
   443→    } else {
   444→      props.onCommonSource(layer, source ?? { type: "none" });
   445→    }
   446→  };
   447→
   448→  const newCodeFile = async () => {
   449→    const name = window.prompt(`New ${layer} code file name (without .tsx)`, `custom-${layer}`);
   450→    if (!name) return;
   451→    const path = `${layer}/${name.replace(/[^\w-]/g, "-").replace(/\.tsx$/, "")}.tsx`;
   452→    const template =
   453→      layer === "bg"
   454→        ? `import { GridLines, Vignette, Aurora } from "stagesmith";
   455→
   456→export default function ${name.replace(/[^\w]/g, "") || "Bg"}() {
   457→  return (
   458→    <>
   459→      <GridLines />
   460→      <Vignette strength={0.3} />
   461→    </>
   462→  );
   463→}
   464→`
   465→        : `import { CornerMarks, SlideNumber } from "stagesmith";
   466→
   467→export default function ${name.replace(/[^\w]/g, "") || "Overlay"}() {
   468→  return (
   469→    <>
   470→      <CornerMarks inset={20} />
   471→      <SlideNumber format="{n} / {total}" />
   472→    </>
   473→  );
   474→}
   475→`;
   476→    try {
   477→      await api.saveFile(deckId, path, template);
   478→      setSource({ type: "code", path });
   479→      toast({ title: `Created ${path}` });
   480→    } catch (e) {
   481→      toast({ title: "Could not create file", description: (e as Error).message, variant: "destructive" });
   482→    }
   483→  };
   484→
   485→  return (
   486→    <div style={{ height: "100%", display: "flex", flexDirection: "column" }}>
   487→      <div className="ss-srcbar">
   488→        <span className="ss-label" style={{ flex: "none" }}>{layer === "bg" ? "Background" : "Overlay"}</span>
   489→        <div className="ss-mode-tabs" style={{ flex: "none" }}>
   490→          <button className={`ss-mode-tab ${scope === "common" ? "on" : ""}`} onClick={() => setScope("common")}>
   491→            Common
   492→          </button>
   493→          <button
   494→            className={`ss-mode-tab ${scope === "override" ? "on" : ""}`}
   495→            disabled={!sel}
   496→            title={sel ? "Override for this slide only" : "Select a slide first"}
   497→            onClick={() => setScope("override")}
   498→          >
   499→            Override {sel ? `· ${sel.label.slice(0, 18)}` : ""}
   500→          </button>
   501→        </div>
   502→        <span className="ss-divider" />
   503→        <Select
   504→          value={effective?.type ?? "none"}
   505→          onValueChange={(v) => {
   506→            if (v === "none") setSource(undefined);
   507→            else if (v === "code") setSource({ type: "code", path: dirFiles[0] ?? `${layer}/common.tsx` });
   508→            else if (v === "image") setSource({ type: "image", path: images[0] ?? "assets/placeholder.png" });
   509→          }}
   510→        >
   511→          <SelectTrigger className="ss-input" style={{ height: 28, width: 150, fontSize: 12 }}>
   512→            <SelectValue />
   513→          </SelectTrigger>
   514→          <SelectContent style={{ background: "var(--ss-panel2)", borderColor: "var(--ss-line-strong)", color: "var(--ss-ink)" }}>
   515→            <SelectItem value="none">None</SelectItem>
   516→            <SelectItem value="code">Code (.tsx)</SelectItem>
   517→            <SelectItem value="image" disabled={images.length === 0}>Image</SelectItem>
   518→          </SelectContent>
   519→        </Select>
   520→
   521→        {effective?.type === "code" && (
   522→          <>
   523→            <Select
   524→              value={codePath ?? ""}
   525→              onValueChange={(v) => setSource({ type: "code", path: v })}
   526→            >
   527→              <SelectTrigger className="ss-input" style={{ height: 28, width: 210, fontSize: 12 }}>
   528→                <SelectValue />
   529→              </SelectTrigger>
   530→              <SelectContent style={{ background: "var(--ss-panel2)", borderColor: "var(--ss-line-strong)", color: "var(--ss-ink)" }}>
   531→                {dirFiles.map((p) => (

   541→
   542→        {effective?.type === "image" && (

   550→                  <SelectItem key={p} value={p}>{p}</SelectItem>
   551→                ))}
   552→              </SelectContent>
   553→            </Select>
   554→            {layer === "bg" ? <ImageBgFields source={effective} onChange={setSource} /> : <ImageOverlayFields source={effective} onChange={setSource} />}

   556→        )}

   558→        <span style={{ font: "500 11px var(--ss-mono)", color: "var(--ss-faint)" }}>

   562→      <div className="ss-cm" style={{ flex: 1 }}>

   565→            value={props.buffers[codePath] ?? ""}

   571→        ) : (
   572→          <EmptyPane
   573→            text={
   574→              effective?.type === "image"

   576→                : "No code source — pick Code + a file, or create one"
   577→            }
   578→          />
   579→        )}
   580→      </div>
   581→    </div>
   582→  );

   585→function ImageBgFields({ source, onChange }: { source: Extract<Source, { type: "image" }>; onChange: (s: Source) => void }) {
   586→  const num = (label: string, key: "blur" | "dim" | "opacity" | "zoom", step = 0.1) => (
   587→    <label key={key} style={{ display: "inline-flex", alignItems: "center", gap: 6, font: "500 11px var(--ss-mono)", color: "var(--ss-mut)" }}>
   588→      {label}
   589→      <input
   590→        type="number"
   591→        step={step}
   592→        min={0}
   593→        max={key === "dim" || key === "opacity" ? 1 : 10}
   594→        value={source[key] ?? 0}
   595→        onChange={(e) => onChange({ ...source, [key]: Number(e.target.value) })}
   596→        style={{ width: 58, height: 26, padding: "0 6px", background: "var(--ss-panel)", color: "var(--ss-ink)", border: "1px solid var(--ss-line-strong)", borderRadius: 6, font: "400 11px var(--ss-mono)", outline: "none" }}
   597→      />
   598→    </label>
   599→  );
   600→  return (
   601→    <>
   602→      <Select value={source.fit ?? "cover"} onValueChange={(v) => onChange({ ...source, fit: v as "cover" | "contain" })}>
   603→        <SelectTrigger className="ss-input" style={{ height: 28, width: 110, fontSize: 12 }}>
   604→          <SelectValue />
   605→        </SelectTrigger>
   606→        <SelectContent style={{ background: "var(--ss-panel2)", borderColor: "var(--ss-line-strong)", color: "var(--ss-ink)" }}>
   607→          <SelectItem value="cover">cover</SelectItem>
   608→          <SelectItem value="contain">contain</SelectItem>
   609→        </SelectContent>
   610→      </Select>
   611→      {num("blur", "blur")}
   612→      {num("dim", "dim")}
   613→      {num("op", "opacity")}
   614→    </>
   615→  );
   616→}

   647→function CodeFileList({
   648→  deckId,
   649→  dir,
   650→  version,
   651→  onOpenFile,
   652→}: {
   653→  deckId: string;
   654→  dir: "overlay" | "bg";
   655→  version: string;
   656→  onOpenFile?: (path: string) => void;
   657→}) {
   658→  const { toast } = useToast();
   659→  const [files, setFiles] = useState<FileRow[] | null>(null);
   660→
   661→  const refresh = useCallback(async () => {
   662→    try {
   663→      setFiles(await api.listCodeFiles(deckId, dir));
   664→    } catch {
   665→      setFiles([]);
   666→    }
   667→  }, [deckId, dir]);
   668→
   669→  useEffect(() => {
   670→    refresh();
   671→  }, [refresh, version]);
   672→
   673→  const upload = async (fileList: FileList | File[]) => {
   674→    const list = Array.from(fileList);
   675→    if (!list.length) return;
   676→    const nonTsx = list.filter((f) => !f.name.endsWith(".tsx"));
   677→    if (nonTsx.length) {
   678→      toast({ title: "Only .tsx files allowed", description: nonTsx.map((f) => f.name).join(", "), variant: "destructive" });
   679→      return;
   680→    }
   681→    try {
   682→      await api.uploadCodeFile(deckId, dir, list);
   683→      await refresh();
   684→      toast({ title: `${list.length} file${list.length > 1 ? "s" : ""} uploaded into ${dir}/` });
   685→    } catch (e) {
   686→      toast({ title: "Upload failed", description: (e as Error).message, variant: "destructive" });
   687→    }
   688→  };
   689→
   690→  const rows = files ?? [];
   691→
   692→  return (
   693→    <div>
   694→      <div style={{ display: "flex", gap: 10, alignItems: "center", marginBottom: 14 }}>
   695→        <label className="ss-btn" style={{ cursor: "pointer" }}>
   696→          <Upload size={14} /> Upload .tsx into {dir}/
   697→          <input
   698→            type="file"
   699→            multiple
   700→            accept=".tsx"
   701→            style={{ display: "none" }}
   702→            onChange={(e) => {
   703→              if (e.target.files) upload(e.target.files);
   704→              e.target.value = "";
   705→            }}
   706→          />
   707→        </label>
   708→        <span className="ss-label">Upload React components directly into {dir}/</span>
   709→      </div>
   710→      {rows.length === 0 ? (
   711→        <EmptyPane text={files === null ? `Loading ${dir} files…` : `No ${dir} files yet — upload a .tsx component above`} />
   712→      ) : (
   713→        <div style={{ display: "flex", flexDirection: "column", gap: 4 }}>
   714→          {rows.map((f) => {
   715→            const name = f.path.replace(`${dir}/`, "");
   716→            return (
   717→              <div
   718→                key={f.path}
   719→                style={{
   720→                  display: "flex",
   721→                  alignItems: "center",
   722→                  gap: 8,
   723→                  padding: "6px 8px",
   724→                  borderRadius: 4,
   725→                  background: "var(--ss-surface2, rgba(255,255,255,0.04))",
   726→                }}
   727→              >
   728→                <span
   729→                  style={{
   730→                    flex: 1,
   731→                    font: "500 12px var(--ss-mono)",
   732→                    color: "var(--ss-fg)",
   733→                    overflow: "hidden",
   734→                    textOverflow: "ellipsis",
   735→                    whiteSpace: "nowrap",
   736→                  }}
   737→                  title={f.path}
   738→                >
   739→                  {name}
   740→                </span>
   741→                {onOpenFile && (
   742→                  <button
   743→                    className="ss-btn"
   744→                    style={{ height: 24, padding: "0 8px", fontSize: 11 }}
   745→                    title="Open in editor"
   746→                    onClick={() => onOpenFile(f.path)}
   747→                  >
   748→                    <Braces size={11} /> open
   749→                  </button>
   750→                )}
   751→                <button
   752→                  className="ss-btn"
   753→                  style={{ height: 24, padding: "0 8px", fontSize: 11 }}
   754→                  title="Copy import path"
   755→                  onClick={async () => {
   756→                    const ok = await copyText(`import './${f.path}';`);
   757→                    toast(ok ? { title: "Path copied" } : { title: "Clipboard blocked", variant: "destructive" });
   758→                  }}
   759→                >
   760→                  <Copy size={11} /> path
   761→                </button>
   762→                <button
   763→                  className="ss-btn"
   764→                  style={{ height: 24, padding: "0 8px", fontSize: 11, color: "var(--ss-err)" }}
   765→                  title="Delete file"
   766→                  onClick={async () => {
   767→                    try {
   768→                      await api.deleteCodeFile(deckId, f.path);
   769→                      refresh();
   770→                    } catch (e) {
   771→                      toast({ title: "Delete failed", description: (e as Error).message, variant: "destructive" });
   772→                    }
   773→                  }}
   774→                >
   775→                  <Trash2 size={11} />
   776→                </button>
   777→              </div>
   778→            );
   779→          })}
   780→        </div>
   781→      )}
   782→    </div>
   783→  );
